def Millennium.pair_encoding : {α β : Type} →
  Computability.FinEncoding α → Computability.FinEncoding β → Computability.FinEncoding (α × β) :=
fun {α β} ea eb =>
  { Γ := Millennium.pair_symbol✝ ea eb,
    encode := fun p =>
      List.map Sum.inl (ea.encode p.1) ++
        Sum.inr Option.none :: List.map (fun b => Sum.inr (Option.some b)) (eb.encode p.2),
    decode := fun l =>
      match Millennium.parse_pair✝ l with
      | Option.some (a_list, b_list) =>
        match ea.decode a_list, eb.decode b_list with
        | Option.some a, Option.some b => Option.some (a, b)
        | x, x_1 => Option.none
      | Option.none => Option.none,
    decode_encode := ⋯, ΓFin := inferInstance }
